function sequence2(d[1..n]): k, array[1..k]
  array j, F[0..n]

  for i ← 0 to n do
    j[i] ← 0
    F[i] ← i
    initialize set[i]
 
  for i ← 1 to n do
    k ← find(min(n, d[i]))
    m ← F[k]
    if m ≠ 0 then
      j[m] ← i
      l ← find(m - 1)
      F[k] ← F[l]
      merge(k, l)

  k ← 0
  for i ← 1 to n do
    if j[i] > 0 then
      k ← k + 1
      j[k] ← j[i]

  return k, j[1..k]
