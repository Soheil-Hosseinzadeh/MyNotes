function sequence(d[0..n]): k, array[1..k]
  array j[0..n]

  d[0] ← j[0] ← 0
  k ← j[1] ← 1

  for i ← 2 to n do
    r ← k
    while d[j[r]] > max(d[i], r) do r ← r - 1
    if d[i] > r then
      for m ← k step-1 to r + 1 do j[m + 1] ← j[m]
      j[r + 1] ← i
      k ← k + 1
      
  return k, j[1..k]
